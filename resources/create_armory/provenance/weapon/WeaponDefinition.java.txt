package net.createarmory.weapon;

import net.createarmory.init.CreatearmoryModItems;
import net.minecraft.world.item.Item;

public enum WeaponDefinition {
    M4("m4", "m4texture", Feed.STANAG, 0, 6, 3, 40, true, 1, 9, 1, .45f, 7),
    M16("msixteen", "m4texture", Feed.STANAG, 0, 6, 4, 40, false, 1, 9, .8f, .4f, 7),
    HK416("foursixteen", "m4texture", Feed.STANAG, 0, 6, 3, 35, true, 1, 9, .8f, .4f, 7),
    MAC10("macten", "mazctexture", Feed.MAC, 0, 4, 2, 35, true, 1, 7, 2, .55f, 3),
    BARRET("barret", "barrettexture", Feed.BARRET, 0, 14, 15, 50, false, 1, 16, .3f, 1.5f, 3),
    BOLT_ACTION_SNIPER("boltactionsniper", "barrettexture", Feed.BARRET, 0, 18, 25, 60, false, 1, 16, .15f, 1.8f, 3),
    PIPE("ppistol", "ppistolfix", Feed.DIRECT, 1, 4, 10, 35, false, 1, 7, 3, .8f, 0),
    DOUBLE_BARREL("db", "dbtexture", Feed.SHELL, 2, 2, 10, 45, false, 8, 6, 6, 1.5f, 0),
    JUNK("junkgun", "jgtexture", Feed.DIRECT, 1, 4, 12, 40, false, 3, 7, 5, 1, 0),
    MINIGUN("mgun", "mguntexture", Feed.DRUM, 0, 4, 2, 70, true, 1, 8, 3, .35f, 0),
    UPGRADED_MINIGUN("upgradedminigun", "mguntexture", Feed.DRUM, 0, 4, 1, 70, true, 1, 8, 2, .3f, 0),
    RPG("rpg", "rpgtexture", Feed.ROCKET, 1, 0, 20, 60, false, 1, 3, .5f, 2, 0);

    public enum Feed {
        STANAG(30), MAC(30), BARRET(5), DRUM(100), DIRECT(1), SHELL(2), ROCKET(1);
        public final int capacity;
        Feed(int capacity) { this.capacity = capacity; }
        public Item ammo() {
            return switch (this) {
                case STANAG -> CreatearmoryModItems.FIVE_FIVE_SIX.get();
                case BARRET -> CreatearmoryModItems.FIFTY_CAL.get();
                case SHELL -> CreatearmoryModItems.SHOTGUN_SHELL.get();
                case ROCKET -> CreatearmoryModItems.RPG_ROCKET.get();
                default -> CreatearmoryModItems.NINE_MM.get();
            };
        }
        public boolean detachable() { return ordinal() <= DRUM.ordinal(); }
    }

    public final String model, texture;
    public final Feed feed;
    public final int directCapacity, interval, reloadTicks, pellets, slots;
    public final float damage, speed, spread, recoil;
    public final boolean automatic;

    WeaponDefinition(String model, String texture, Feed feed, int directCapacity, float damage,
                     int interval, int reloadTicks, boolean automatic, int pellets, float speed,
                     float spread, float recoil, int slots) {
        this.model = model; this.texture = texture; this.feed = feed; this.directCapacity = directCapacity;
        this.damage = damage; this.interval = interval; this.reloadTicks = reloadTicks;
        this.automatic = automatic; this.pellets = pellets; this.speed = speed;
        this.spread = spread; this.recoil = recoil; this.slots = slots;
    }

    public boolean supports(AttachmentItem attachment) {
        return (slots & (1 << attachment.slot.ordinal())) != 0;
    }

    public boolean canAim() { return this != MINIGUN && this != UPGRADED_MINIGUN; }
}
