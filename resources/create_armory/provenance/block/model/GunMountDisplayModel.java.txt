package net.createarmory.block.model;

import software.bernie.geckolib.model.GeoModel;

import net.minecraft.resources.ResourceLocation;

import net.createarmory.block.display.GunMountDisplayItem;

public class GunMountDisplayModel extends GeoModel<GunMountDisplayItem> {
	@Override
	public ResourceLocation getAnimationResource(GunMountDisplayItem animatable) {
		return new ResourceLocation("createarmory", "animations/gunmount.animation.json");
	}

	@Override
	public ResourceLocation getModelResource(GunMountDisplayItem animatable) {
		return new ResourceLocation("createarmory", "geo/gunmount.geo.json");
	}

	@Override
	public ResourceLocation getTextureResource(GunMountDisplayItem entity) {
		return new ResourceLocation("createarmory", "textures/block/mounttexture.png");
	}
}
