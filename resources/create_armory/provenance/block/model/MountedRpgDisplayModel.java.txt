package net.createarmory.block.model;

import software.bernie.geckolib.model.GeoModel;

import net.minecraft.resources.ResourceLocation;

import net.createarmory.block.display.MountedRpgDisplayItem;

public class MountedRpgDisplayModel extends GeoModel<MountedRpgDisplayItem> {
	@Override
	public ResourceLocation getAnimationResource(MountedRpgDisplayItem animatable) {
		return new ResourceLocation("createarmory", "animations/mountedrpg.animation.json");
	}

	@Override
	public ResourceLocation getModelResource(MountedRpgDisplayItem animatable) {
		return new ResourceLocation("createarmory", "geo/mountedrpg.geo.json");
	}

	@Override
	public ResourceLocation getTextureResource(MountedRpgDisplayItem entity) {
		return new ResourceLocation("createarmory", "textures/block/mountedrpgtexture.png");
	}
}
