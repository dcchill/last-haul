package net.createarmory.block.model;

import software.bernie.geckolib.model.GeoModel;

import net.minecraft.resources.ResourceLocation;

import net.createarmory.block.display.MountedMinigunDisplayItem;

public class MountedMinigunDisplayModel extends GeoModel<MountedMinigunDisplayItem> {
	@Override
	public ResourceLocation getAnimationResource(MountedMinigunDisplayItem animatable) {
		return new ResourceLocation("createarmory", "animations/mountedminigun.animation.json");
	}

	@Override
	public ResourceLocation getModelResource(MountedMinigunDisplayItem animatable) {
		return new ResourceLocation("createarmory", "geo/mountedminigun.geo.json");
	}

	@Override
	public ResourceLocation getTextureResource(MountedMinigunDisplayItem entity) {
		return new ResourceLocation("createarmory", "textures/block/mountedmgun.png");
	}
}
