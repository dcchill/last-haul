package net.createarmory.block.model;

import software.bernie.geckolib.model.GeoModel;

import net.minecraft.resources.ResourceLocation;

import net.createarmory.block.entity.MountedMinigunTileEntity;

public class MountedMinigunBlockModel extends GeoModel<MountedMinigunTileEntity> {
	@Override
	public ResourceLocation getAnimationResource(MountedMinigunTileEntity animatable) {
		return new ResourceLocation("createarmory", "animations/mountedminigun.animation.json");
	}

	@Override
	public ResourceLocation getModelResource(MountedMinigunTileEntity animatable) {
		return new ResourceLocation("createarmory", "geo/mountedminigun.geo.json");
	}

	@Override
	public ResourceLocation getTextureResource(MountedMinigunTileEntity entity) {
		return new ResourceLocation("createarmory", "textures/block/mountedmgun.png");
	}
}
