package net.createarmory.block.renderer;

import software.bernie.geckolib.renderer.GeoBlockRenderer;

import net.minecraft.resources.ResourceLocation;
import net.minecraft.client.renderer.RenderType;
import net.minecraft.client.renderer.MultiBufferSource;

import net.createarmory.block.model.MountedMinigunBlockModel;
import net.createarmory.block.entity.MountedMinigunTileEntity;

public class MountedMinigunTileRenderer extends GeoBlockRenderer<MountedMinigunTileEntity> {
	public MountedMinigunTileRenderer() {
		super(new MountedMinigunBlockModel());
	}

	@Override
	public RenderType getRenderType(MountedMinigunTileEntity animatable, ResourceLocation texture, MultiBufferSource bufferSource, float partialTick) {
		return RenderType.entityTranslucent(getTextureLocation(animatable));
	}
}
