package net.createarmory.block.renderer;

import software.bernie.geckolib.renderer.GeoItemRenderer;

import net.minecraft.resources.ResourceLocation;
import net.minecraft.client.renderer.RenderType;
import net.minecraft.client.renderer.MultiBufferSource;

import net.createarmory.block.model.GunMountDisplayModel;
import net.createarmory.block.display.GunMountDisplayItem;

public class GunMountDisplayItemRenderer extends GeoItemRenderer<GunMountDisplayItem> {
	public GunMountDisplayItemRenderer() {
		super(new GunMountDisplayModel());
	}

	@Override
	public RenderType getRenderType(GunMountDisplayItem animatable, ResourceLocation texture, MultiBufferSource bufferSource, float partialTick) {
		return RenderType.entityTranslucent(getTextureLocation(animatable));
	}
}
