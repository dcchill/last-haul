package net.createarmory.block.renderer;

import software.bernie.geckolib.renderer.GeoBlockRenderer;

import net.minecraft.resources.ResourceLocation;
import net.minecraft.client.renderer.RenderType;
import net.minecraft.client.renderer.MultiBufferSource;

import net.createarmory.block.model.MountedRpgBlockModel;
import net.createarmory.block.entity.MountedRpgTileEntity;

public class MountedRpgTileRenderer extends GeoBlockRenderer<MountedRpgTileEntity> {
	public MountedRpgTileRenderer() {
		super(new MountedRpgBlockModel());
	}

	@Override
	public RenderType getRenderType(MountedRpgTileEntity animatable, ResourceLocation texture, MultiBufferSource bufferSource, float partialTick) {
		return RenderType.entityTranslucent(getTextureLocation(animatable));
	}
}
