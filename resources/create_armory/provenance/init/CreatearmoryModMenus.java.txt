package net.createarmory.init;

import net.createarmory.CreatearmoryMod;
import net.createarmory.menu.MagazineMenu;
import net.minecraft.world.inventory.MenuType;
import net.minecraftforge.common.extensions.IForgeMenuType;
import net.minecraftforge.registries.*;

public final class CreatearmoryModMenus {
    public static final DeferredRegister<MenuType<?>> REGISTRY = DeferredRegister.create(ForgeRegistries.MENU_TYPES, CreatearmoryMod.MODID);
    public static final RegistryObject<MenuType<MagazineMenu>> MAGAZINE = REGISTRY.register("magazine",
        () -> IForgeMenuType.create((id, inventory, buffer) -> new MagazineMenu(id, inventory, buffer.readVarInt())));
}
